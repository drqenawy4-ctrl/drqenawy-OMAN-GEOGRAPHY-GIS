function openLocation(name, lat, lon) { const osm = `https://www.openstreetmap.org/?mlat=${lat}&mlon=${lon}#map=16/${lat}/${lon}`; const google = `https://www.google.com/maps/search/?api=1&query=${lat}%2C${lon}`; document.getElementById("map").scrollIntoView({ behavior: "smooth", block: "center" }); if (window.map && typeof window.map.setView === "function") { window.map.setView([lat, lon], 16); } window.open(osm, "_blank", "noopener"); // رابط Google Maps بديل مستقل const link = document.getElementById("googleFallback"); if (link) { link.href = google; link.textContent = "فتح " + name + " في Google Maps"; link.style.display = "inline-block"; } }
